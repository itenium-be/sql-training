Which country has "scored" the most owngoals? Also select how many.

SELECT TOP 1 c.name, COUNT(*) AS owngoals
FROM goals g
JOIN teams t ON t.id = g.team_id
JOIN countries c ON c.id = t.country_id
WHERE g.owngoal = 1
GROUP BY c.name
ORDER BY owngoals DESC
